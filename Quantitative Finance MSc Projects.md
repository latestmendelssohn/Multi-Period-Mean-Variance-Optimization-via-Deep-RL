`# 1. THE RAG ENGINE (LangChain, Chroma, PyPDF)`  
`# ==========================================`  
`class FinancialRAGAgent:`  
    `def __init__(self, api_key: str):`  
        `os.environ["GOOGLE_API_KEY"] = api_key`  
        `# Initialize Gemini LLM and Embeddings`  
        `self.llm = ChatGoogleGenerativeAI(model="gemini-1.5-pro", temperature=0.0)`  
        `self.embeddings = GoogleGenerativeAIEmbeddings(model="models/embedding-001")`  
        `self.vector_store = None`

    `def ingest_pdf(self, pdf_path: str):`  
        `"""Uses PyPDF to load the document and ChromaDB to vectorize it."""`  
        `print(f"Loading PDF: {pdf_path}")`  
        `loader = PyPDFLoader(pdf_path)`  
        `documents = loader.load()`  
          
        `# Split text into manageable chunks for the LLM`  
        `text_splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=100)`  
        `chunks = text_splitter.split_documents(documents)`  
          
        `# Store in local ChromaDB vector database`  
        `self.vector_store = Chroma.from_documents(chunks, self.embeddings)`  
        `print(f"Successfully vectorized {len(chunks)} chunks into ChromaDB.")`

    `def extract_financial_constraints(self, asset_list: list) -> dict:`  
        `"""Queries the vector database and forces Gemini to output JSON bounds."""`  
        `if not self.vector_store:`  
            `raise ValueError("No documents ingested into ChromaDB yet.")`

        `query = f"What are the maximum portfolio weight limits for: {', '.join(asset_list)}?"`  
        `retrieved_docs = self.vector_store.similarity_search(query, k=4)`  
        `context = "\n".join([doc.page_content for doc in retrieved_docs])`  
          
        `prompt = f"""`  
        `You are an elite quantitative risk manager. Based strictly on the provided context,`   
        `extract the maximum allowed portfolio weight (as a decimal) for the requested assets.`  
          
        `CONTEXT: {context}`  
        `ASSETS: {asset_list}`  
          
        `OUTPUT: Return ONLY a valid JSON dictionary where keys are asset names and values are`   
        `the decimal maximums. Example: {{"Tech": 0.20, "Bonds": 0.50}}`  
        `"""`  
          
        `response = self.llm.invoke(prompt)`  
        `try:`  
            ````clean_json = response.content.replace("```json", "").replace("```", "").strip()````  
            `return json.loads(clean_json)`  
        `except Exception as e:`  
            `print(f"Failed to parse LLM output: {e}")`  
            `return {}`

`# ==========================================`  
`# 2. THE FINANCIAL ENGINE (Pandas & SciPy)`  
`# ==========================================`  
`def optimize_portfolio(expected_returns: pd.Series, cov_matrix: pd.DataFrame, ai_bounds: dict):`  
    `"""Minimizes portfolio volatility subject to AI-extracted RAG constraints."""`  
    `n_assets = len(expected_returns)`  
    `asset_names = expected_returns.index.tolist()`  
      
    `def objective(weights):`  
        `return np.sqrt(weights.T @ cov_matrix.values @ weights)`  
      
    `# Sum of weights must equal 100%`  
    `constraints = [{'type': 'eq', 'fun': lambda w: np.sum(w) - 1.0}]`  
      
    `# Default bounds (0 to 100%)`  
    `bounds = [[0.0, 1.0] for _ in range(n_assets)]`  
      
    `# Apply strict AI RAG constraints`  
    `for i, asset in enumerate(asset_names):`  
        `if asset in ai_bounds:`  
            `bounds[i][1] = ai_bounds[asset] # Update max bound`  
              
    `# Solve using Sequential Least Squares Programming`  
    `initial_guess = np.ones(n_assets) / n_assets`  
    `result = minimize(objective, initial_guess, method='SLSQP', bounds=bounds, constraints=constraints)`  
      
    `if result.success:`  
        `return pd.Series(result.x, index=asset_names).round(4).to_dict()`  
    `else:`  
        `raise Exception("Optimization failed to converge under given AI constraints.")`

`# ==========================================`  
`# 1. TIME-SERIES DATA ENGINEERING`  
`# ==========================================`  
`def generate_atm_data(days=1000):`  
    `"""Generates realistic ATM cash withdrawal data with weekly/monthly seasonality."""`  
    `np.random.seed(42)`  
    `dates = pd.date_range(start='2021-01-01', periods=days)`  
      
    `# Base demand + noise`  
    `demand = np.random.normal(loc=50000, scale=8000, size=days)`  
      
    `df = pd.DataFrame({'Date': dates, 'Cash_Demand': demand})`  
    `df['DayOfWeek'] = df['Date'].dt.dayofweek`  
    `df['DayOfMonth'] = df['Date'].dt.day`  
      
    `# Seasonality: Fridays (4) and Saturdays (5) have 30% higher demand`  
    `df.loc[df['DayOfWeek'].isin([4, 5]), 'Cash_Demand'] *= 1.3`  
      
    `# Seasonality: Paydays (1st and last day of month) spike by 40%`  
    `df.loc[df['DayOfMonth'].isin([1, 28, 29, 30, 31]), 'Cash_Demand'] *= 1.4`  
      
    `return df.set_index('Date')`

`def create_lag_features(df, target_col):`  
    `"""Engineers auto-regressive features required for Time-Series ML."""`  
    `print("Engineering Temporal Features and Lags...")`  
    `df_feat = df.copy()`  
    `# Lag features (What was the demand yesterday? A week ago?)`  
    `for lag in [1, 2, 7, 14, 30]:`  
        `df_feat[f'Lag_{lag}'] = df_feat[target_col].shift(lag)`  
          
    `# Rolling features (Momentum of cash demand)`  
    `df_feat['Rolling_Mean_7'] = df_feat[target_col].shift(1).rolling(window=7).mean()`  
    `df_feat['Rolling_Std_7'] = df_feat[target_col].shift(1).rolling(window=7).std()`  
      
    `return df_feat.dropna()`

`# ==========================================`  
`# 2. MACHINE LEARNING FORECASTING (XGBoost)`  
`# ==========================================`  
`def train_forecasting_model(df, target_col):`  
    `"""Trains an XGBoost model using rigorous Time-Series cross-validation."""`  
    `features = [col for col in df.columns if col not in [target_col]]`  
      
    `# Chronological Split (No data leakage from the future)`  
    `train_size = int(len(df) * 0.8)`  
    `train, test = df.iloc[:train_size], df.iloc[train_size:]`  
      
    `X_train, y_train = train[features], train[target_col]`  
    `X_test, y_test = test[features], test[target_col]`  
      
    `print("Training XGBoost Regressor...")`  
    `model = xgb.XGBRegressor(n_estimators=200, learning_rate=0.05, max_depth=4, random_state=42)`  
    `model.fit(X_train, y_train)`  
      
    `predictions = model.predict(X_test)`  
    `rmse = np.sqrt(mean_squared_error(y_test, predictions))`  
    `print(f"Out-of-Sample RMSE: ${rmse:,.2f}")`  
      
    `return model, features, rmse, test`

`# ==========================================`  
`# 3. STOCHASTIC OPTIMIZATION ENGINE`  
`# ==========================================`  
`def optimize_cash_load(forecasted_mean, forecast_rmse, holding_cost_rate, stockout_penalty):`  
    `"""`  
    `Mathematical Inventory Optimization using the Newsvendor Model.`  
    `Calculates the exact percentile of the distribution to target based on cost ratios.`  
    `"""`  
    `# Critical Ratio (Service Level)`  
    `# If stockout penalty is huge, critical ratio approaches 0.99 (stock a lot of cash)`  
    `critical_ratio = stockout_penalty / (stockout_penalty + holding_cost_rate)`  
      
    `# Calculate optimal cash reload using Inverse CDF (Percent Point Function)`  
    `optimal_cash = norm.ppf(critical_ratio, loc=forecasted_mean, scale=forecast_rmse)`  
      
    `return optimal_cash, critical_ratio`

`# ==========================================`  
`if __name__ == "__main__":`

    `# 2. Train ML Model`  
    `model, feature_cols, model_rmse, test_data = train_forecasting_model(ml_data, 'Cash_Demand')`  
      
    `# 3. Simulate Operations for Tomorrow`  
    `tomorrow_features = test_data.iloc[-1:][feature_cols]`  
    `expected_demand = model.predict(tomorrow_features)[0]`  
      
    `# Financial Parameters (Daily rates)`  
    `# Example: 5% annual interest rate / 365 = ~0.00013 daily holding cost`  
    `h_cost = 0.00013`    
    `# Penalty for empty ATM (loss of fee, angry customer risk) = $2.00 per dollar short`

        `forecasted_mean=expected_demand,`   
        `forecast_rmse=model_rmse,`   
        `holding_cost_rate=h_cost,`   
        `stockout_penalty=p_cost`  
    `)`  
      
    `print("\n" + "="*45)`

### **The Architecture Code (The Implementation Blueprint)**

Because this architecture spans multiple domains, this script provides the structural masterclass. It contains the PyTorch Graph Neural Network for demand forecasting, the custom Gymnasium environment for the armored truck logistics, and the PPO training loop.

*Requirements:* pip install torch torch\_geometric gymnasium stable-baselines3 networkx numpy pandas

`import numpy as np`  
`import networkx as nx`  
`import torch`  
`import torch.nn as nn`  
`import torch.nn.functional as F`  
`from torch_geometric.nn import GCNConv`  
`import gymnasium as gym`  
`from gymnasium import spaces`  
`from stable_baselines3 import PPO`

`# ==========================================`  
`# 1. SPATIO-TEMPORAL GNN (Demand Forecasting)`  
`# ==========================================`  
`class SpatioTemporalGNN(torch.nn.Module):`  
    `"""`  
    `Takes a geographic graph of ATMs and temporal lag features to predict`   
    `simultaneous cash demand across the entire network, accounting for spillover.`  
    `"""`  
    `def __init__(self, num_node_features, hidden_channels):`  
        `super(SpatioTemporalGNN, self).__init__()`  
        `# Graph Convolutions to learn spatial relationships (ATMs on the same street)`  
        `self.conv1 = GCNConv(num_node_features, hidden_channels)`  
        `self.conv2 = GCNConv(hidden_channels, hidden_channels)`  
          
        `# Temporal Fully Connected layer for final demand regression`  
        `self.fc = nn.Linear(hidden_channels, 1)`

    `def forward(self, x, edge_index):`  
        `# Pass node features (historical demand) through the graph edges`  
        `x = self.conv1(x, edge_index)`  
        `x = F.relu(x)`  
        `x = F.dropout(x, p=0.2, training=self.training)`  
          
        `x = self.conv2(x, edge_index)`  
        `x = F.relu(x)`  
          
        `# Output predicted cash demand for every node (ATM) simultaneously`  
        `out = self.fc(x)`  
        `return out`

`# ==========================================`  
`# 2. MARKOV DECISION PROCESS (Gymnasium Env)`  
`# ==========================================`  
`class ATMRoutingEnv(gym.Env):`  
    `"""`  
    `Custom MDP Environment simulating non-linear routing costs and cash inventory.`  
    `The RL Agent must decide which ATMs to refill and by how much, daily.`  
    `"""`  
    `def __init__(self, num_atms=10, max_capacity=100000):`  
        `super(ATMRoutingEnv, self).__init__()`  
        `self.num_atms = num_atms`  
        `self.max_capacity = max_capacity`  
          
        `# Action: What % of max_capacity to load into each of the N ATMs today`  
        `self.action_space = spaces.Box(low=0, high=1, shape=(self.num_atms,), dtype=np.float32)`  
          
        `# State: Current cash levels at each ATM + Day of the week`  
        `self.observation_space = spaces.Box(low=0, high=self.max_capacity, shape=(self.num_atms + 1,), dtype=np.float32)`  
          
        `# Economics`  
        `self.holding_cost_rate = 0.00015  # Cost of idle capital`  
        `self.stockout_penalty = 2.0       # Massive penalty for empty ATM`  
        `self.truck_dispatch_base_cost = 250.0 # Fixed cost just to roll a truck`  
        `self.cost_per_stop = 35.0         # Marginal cost per ATM refilled`  
          
        `self.state = None`  
        `self.day_step = 0`

    `def reset(self, seed=None):`  
        `self.day_step = 0`  
        `# Initialize ATMs half full`  
        `cash_levels = np.full(self.num_atms, self.max_capacity / 2.0)`  
        `self.state = np.append(cash_levels, self.day_step)`  
        `return np.array(self.state, dtype=np.float32), {}`

    `def step(self, action):`  
        `cash_levels = self.state[:self.num_atms]`  
          
        `# 1. Execute Agent's Actions (Reloading ATMs)`  
        `reload_amounts = action * self.max_capacity`  
        `cash_levels = np.clip(cash_levels + reload_amounts, 0, self.max_capacity)`  
          
        `# Calculate Non-Linear CIT Routing Costs`  
        `num_stops = np.sum(reload_amounts > 0.01) # Number of ATMs visited today`  
        `if num_stops > 0:`  
            `routing_cost = self.truck_dispatch_base_cost + (num_stops * self.cost_per_stop)`  
        `else:`  
            `routing_cost = 0.0`  
              
        `# 2. Simulate True Network Demand (In reality, fetched from GNN)`  
        `actual_demand = np.random.normal(5000, 2000, self.num_atms)`  
          
        `# 3. Calculate Financial Penalties`  
        `post_demand_cash = cash_levels - actual_demand`  
          
        `stockouts = np.minimum(post_demand_cash, 0) * -1`  
        `idle_cash = np.maximum(post_demand_cash, 0)`  
          
        `total_stockout_cost = np.sum(stockouts) * self.stockout_penalty`  
        `total_holding_cost = np.sum(idle_cash) * self.holding_cost_rate`  
          
        `# 4. The Reward Function (Agent must minimize this total loss)`  
        `total_daily_cost = routing_cost + total_stockout_cost + total_holding_cost`  
        `reward = -total_daily_cost`    
          
        `# Update State`  
        `self.day_step += 1`  
        `self.state = np.append(idle_cash, self.day_step % 7) # Track day of week`  
          
        `done = bool(self.day_step >= 365) # End of episode (1 year)`  
          
        `return np.array(self.state, dtype=np.float32), reward, done, False, {}`

`# ==========================================`  
`# 3. MLOps: DISTRIBUTED PPO TRAINING`  
`# ==========================================`  
`if __name__ == "__main__":`  
    `print("Initializing Network Environment...")`  
    `env = ATMRoutingEnv(num_atms=20)`  
      
    `print("Instantiating Deep Proximal Policy Optimization (PPO) Agent...")`  
    `# Multi-Layer Perceptron Policy to learn the complex reward surface`  
    `model = PPO("MlpPolicy", env, verbose=1, learning_rate=3e-4, n_steps=2048, batch_size=64)`  
      
    `print("Commencing Deep Reinforcement Learning (This step requires cluster GPU in prod)...")`  
    `# In a real scenario, this runs for millions of timesteps over several weeks`  
    `model.learn(total_timesteps=100000)`  
      
    `print("\nTraining Complete. Deploying Agent into Simulation...")`  
    `obs, _ = env.reset()`  
    `for _ in range(5):`  
        `action, _states = model.predict(obs, deterministic=True)`  
        `obs, reward, done, _, _ = env.step(action)`  
        `print(f"Agent Action (Reload % per ATM): {np.round(action, 2)}")`  
        `print(f"Daily Operating Reward (Negative Cost): ${reward:,.2f}")`
